function fn_egov_regist_RecentSrchwrdResult(SrchwrdManageId, searchKeyword, excFunction, frm){
	var url = frm.rsm_url.value;
	var param = {
			srchwrdManageId: SrchwrdManageId,
			srchwrdNm: searchKeyword
	};
	$.ajax({
		url: url,
		method: "POST",
		data: param,
		success: function(returnData) {
			if(returnData == 'Success'){
				eval(excFunction);
			}
		},
		error: function() {
			console.error("최근검색어 등록 실패");
		}
	});
}
$(document).ready(function(){
	var ajaxUrl = "<c:url value='/uss/ion/rsm/listRecentSrchwrdResultSerach.do' />";
	var ajaxParam = {
		srchwrdManageId: "SRCMGR_0000000000001"
		,searchKeyword: ""
	};
    $("#searchKeyword").autocomplete({
		source: function(request, response){
			$.ajax({
				url: ajaxUrl
				,contentType: "application/x-www-form-urlennocoded; charset=UTF-8"
				,data: ajaxParam
				,dataType: 'xml'
				,success: function(returnData, status) {
					var items = [];
					$(returnData).find("name").each(function(idx,data) {
						items.push($(data).text());
					});
					response(items);
				}
			});
		},
		minLength : 1,
		select: function(event, ui){
		   $("#searchKeyword").val(this.value);
		}
    });
});
